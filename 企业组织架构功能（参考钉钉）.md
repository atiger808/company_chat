# 企业组织架构功能开发（org 应用）

基于当前 `企联云` 项目技术栈，参考钉钉组织架构设计，新建 `org` 应用实现完整的企业组织架构管理功能。

---

## 📋 一、功能概述

仿照钉钉企业组织架构，实现以下核心功能：

### 1.1 核心功能清单

| 功能模块       | 功能点                                                       | 优先级 |
| -------------- | ------------------------------------------------------------ | ------ |
| **部门管理**   | 多层级树形结构（无限层级）、创建/编辑/删除/合并部门、部门排序、部门编码 | P0     |
| **负责人管理** | 支持多个负责人（正职+副职）、负责人变更通知                  | P0     |
| **员工归属**   | 主部门+兼职部门（多部门归属）、部门内职位、入职日期          | P0     |
| **汇报关系**   | 直属上级、虚线汇报、汇报关系可视化                           | P1     |
| **可见性控制** | 全公司可见/仅本部门/自定义可见/隐藏部门                      | P1     |
| **部门群同步** | 自动创建部门群、成员变动自动同步                             | P1     |
| **组织架构图** | 树形可视化、缩放拖拽、节点详情                               | P1     |
| **分级管理员** | 超级管理员/部门管理员/部门主管/普通员工                      | P1     |
| **搜索导航**   | 全局搜索、面包屑导航、快速定位                               | P0     |
| **批量操作**   | Excel 导入/导出部门与员工                                    | P2     |
| **审计日志**   | 部门变更日志、成员变动日志                                   | P2     |

```python
from django.db import models
from django.conf import settings
from django.utils import timezone


class Department(models.Model):
    """部门模型（多层级树形结构）- 企业级增强版"""
    
    # ===== 基础信息 =====
    name = models.CharField(max_length=100, verbose_name='部门名称')
    code = models.CharField(max_length=50, unique=True, 
                            verbose_name='部门编码',
                            help_text='如：TECH、HR、FINANCE')
    parent = models.ForeignKey('self', null=True, blank=True,
                               on_delete=models.CASCADE,
                               related_name='children',
                               verbose_name='上级部门')
    
    # ===== 层级与排序（新增） =====
    level = models.IntegerField(default=1, verbose_name='层级')
    sort_order = models.IntegerField(default=0, verbose_name='排序号')
    full_path = models.CharField(max_length=500, blank=True,
                                 verbose_name='完整路径',
                                 help_text='如：公司/技术部/后端组')
    
    # ===== 部门属性（新增） =====
    description = models.TextField(blank=True, verbose_name='部门描述')
    department_type = models.CharField(
        max_length=20,
        choices=[
            ('company', '公司'),
            ('department', '部门'),
            ('group', '小组'),
            ('virtual', '虚拟组织')
        ],
        default='department',
        verbose_name='部门类型'
    )
    
    # ===== 负责人（增强：支持多个） =====
    manager = models.ForeignKey(
        settings.AUTH_USER_MODEL,
        on_delete=models.SET_NULL,
        null=True,
        blank=True,
        verbose_name='主负责人',
        related_name='managed_departments'
    )
    deputy_managers = models.ManyToManyField(
        settings.AUTH_USER_MODEL,
        related_name='deputy_managed_departments',
        blank=True,
        verbose_name='副负责人'
    )
    
    # ===== 可见性控制（新增） =====
    visibility = models.CharField(
        max_length=20,
        choices=[
            ('public', '全公司可见'),
            ('department', '仅本部门及子部门可见'),
            ('custom', '自定义可见范围'),
            ('hidden', '隐藏部门（仅管理员可见）')
        ],
        default='public',
        verbose_name='可见范围'
    )
    visible_departments = models.ManyToManyField(
        'self',
        symmetrical=False,
        related_name='visible_by',
        blank=True,
        verbose_name='可见部门列表'
    )
    
    # ===== 部门群关联（新增） =====
    department_group = models.OneToOneField(
        'chat.ChatRoom',
        null=True,
        blank=True,
        on_delete=models.SET_NULL,
        related_name='department',
        verbose_name='部门群聊'
    )
    
    # ===== 自动同步设置（新增） =====
    auto_create_group = models.BooleanField(default=True, 
                                            verbose_name='自动创建部门群')
    auto_sync_members = models.BooleanField(default=True, 
                                            verbose_name='自动同步成员到群')
    
    # ===== 标签与扩展（新增） =====
    tags = models.JSONField(default=list, blank=True, 
                            verbose_name='部门标签')
    custom_fields = models.JSONField(default=dict, blank=True,
                                     verbose_name='自定义字段')
    
    # ===== 状态 =====
    is_active = models.BooleanField(default=True, verbose_name='是否启用')
    
    # ===== 审计字段 =====
    created_by = models.ForeignKey(
        settings.AUTH_USER_MODEL,
        on_delete=models.SET_NULL,
        null=True,
        blank=True,
        related_name='created_departments',
        verbose_name='创建人'
    )
    created_at = models.DateTimeField(auto_now_add=True)
    updated_at = models.DateTimeField(auto_now=True)
    
    class Meta:
        verbose_name = '部门'
        verbose_name_plural = '部门'
        ordering = ['sort_order', 'id']
        indexes = [
            models.Index(fields=['parent', 'is_active']),
            models.Index(fields=['code']),
            models.Index(fields=['level']),
        ]
    
    def __str__(self):
        return self.name
    
    def save(self, *args, **kwargs):
        # 自动计算层级和路径
        if self.parent:
            self.level = self.parent.level + 1
            self.full_path = f'{self.parent.full_path}/{self.name}'
        else:
            self.level = 1
            self.full_path = self.name
        super().save(*args, **kwargs)
    
    def get_all_members(self):
        """获取部门所有成员（包括兼职）"""
        user_ids = UserDepartment.objects.filter(
            department=self
        ).values_list('user_id', flat=True)
        from accounts.models import CustomUser
        return CustomUser.objects.filter(id__in=user_ids)
    
    def get_ancestor_ids(self):
        """获取所有祖先部门ID"""
        ids = []
        current = self.parent
        while current:
            ids.append(current.id)
            current = current.parent
        return ids
    
    def get_descendant_ids(self):
        """获取所有子孙部门ID"""
        ids = []
        children = Department.objects.filter(parent=self)
        for child in children:
            ids.append(child.id)
            ids.extend(child.get_descendant_ids())
        return ids
```

```python
class CustomUser(AbstractUser):
    # ... 保留原有字段 ...
    
    # 保留原有的 department 字段作为主部门（向后兼容）
    # 新增方法支持多部门查询
    
    def get_primary_department(self):
        """获取主部门"""
        primary_rel = self.department_relations.filter(is_primary=True).first()
        if primary_rel:
            return primary_rel.department
        return self.department  # 降级到原有字段
    
    def get_all_departments(self):
        """获取所有部门（包括兼职）"""
        dept_ids = UserDepartment.objects.filter(
            user=self
        ).values_list('department_id', flat=True)
        return Department.objects.filter(id__in=dept_ids)
    
    def get_position_in_department(self, department_id):
        """获取在指定部门的职位"""
        rel = self.department_relations.filter(
            department_id=department_id
        ).first()
        return rel.position if rel else self.position
    
    def get_direct_supervisor(self):
        """获取直属上级"""
        relation = self.report_to_relations.filter(is_direct=True).first()
        return relation.supervisor if relation else None
    
    def get_subordinates(self):
        """获取下属列表"""
        return CustomUser.objects.filter(
            report_to_relations__supervisor=self,
            report_to_relations__is_direct=True
        )
```

### 5.1 分级管理

```python
# org/permissions.py
from rest_framework import permissions

class OrgPermission(permissions.BasePermission):
    """组织架构权限控制"""
    
    def has_permission(self, request, view):
        user = request.user
        
        # 超级管理员：所有权限
        if user.user_type == 'super_admin':
            return True
        
        # 管理员：管理本部门及子部门
        if user.user_type == 'admin':
            return self.can_manage_department(user, view)
        
        # 部门主管：查看本部门成员
        if self.is_department_leader(user):
            return view.action in ['list', 'retrieve', 'members']
        
        # 普通员工：仅查看公开部门
        return view.action in ['list', 'retrieve']
    
    def can_manage_department(self, user, view):
        """检查是否可以管理部门"""
        # 检查用户是否是某个部门的负责人
        return user.managed_departments.exists()
    
    def is_department_leader(self, user):
        """检查是否是部门主管"""
        return user.managed_departments.exists() or \
               user.deputy_managed_departments.exists()
```

```python
def get_visible_departments(user):
    """获取用户可见的部门列表"""
    if user.user_type in ['super_admin', 'admin']:
        return Department.objects.all()

    visible_q = models.Q(visibility='public')
    member_q = models.Q(id__in=user.get_all_departments().values_list('id', flat=True))
    
    # 自定义可见范围
    custom_q = models.Q(visible_departments__in=user.get_all_departments())
    
    return Department.objects.filter(visible_q | member_q | custom_q).distinct()
```

### 6.1 部门群自动创建（`org/signals.py`）

```python
# org/signals.py
from django.db.models.signals import post_save, m2m_changed
from django.dispatch import receiver
from .models import Department, UserDepartment

@receiver(post_save, sender=Department)
def create_department_group(sender, instance, created, **kwargs):
    """部门创建时自动创建群聊"""
    if created and instance.auto_create_group:
        from chat.models import ChatRoom
        
        # 创建部门群聊
        group = ChatRoom.objects.create(
            room_type='group',
            name=f'{instance.name}部门群',
            creator=instance.created_by,
            is_department_group=True
        )
        
        # 添加部门成员
        members = instance.get_all_members()
        group.members.add(*members)
        
        instance.department_group = group
        instance.save(update_fields=['department_group'])

@receiver(m2m_changed, sender=UserDepartment)
def sync_department_group_members(sender, instance, action, **kwargs):
    """员工部门关系变更时同步部门群成员"""
    if action in ['post_add', 'post_remove']:
        department = instance.department
        if department.department_group and department.auto_sync_members:
            members = department.get_all_members()
            department.department_group.members.set(members)
```

## 🔄 八、数据迁移策略

```python
# org/migrations/0001_initial.py
def migrate_existing_departments(apps, schema_editor):
    """迁移现有 Department 数据"""
    OldDepartment = apps.get_model('accounts', 'Department')
    NewDepartment = apps.get_model('org', 'Department')
    UserDepartment = apps.get_model('org', 'UserDepartment')
    CustomUser = apps.get_model('accounts', 'CustomUser')
    
    # 1. 迁移部门数据
    for old_dept in OldDepartment.objects.all():
        new_dept, created = NewDepartment.objects.get_or_create(
            id=old_dept.id,
            defaults={
                'name': old_dept.name,
                'code': old_dept.code or f'DEPT_{old_dept.id}',
                'parent_id': old_dept.parent_id,
                'manager_id': old_dept.manager_id,
                'level': 1,  # 后续重新计算
            }
        )
    
    # 2. 重新计算层级
    for dept in NewDepartment.objects.filter(parent__isnull=True):
        update_department_level(dept, 1)
    
    # 3. 迁移用户部门关系
    for user in CustomUser.objects.filter(department__isnull=False):
        UserDepartment.objects.get_or_create(
            user=user,
            department_id=user.department_id,
            defaults={'is_primary': True, 'position': user.position}
        )

def update_department_level(department, level):
    """递归更新部门层级"""
    department.level = level
    department.save(update_fields=['level'])
    for child in department.children.all():
        update_department_level(child, level + 1)
```
