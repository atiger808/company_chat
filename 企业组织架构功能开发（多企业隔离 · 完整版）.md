# 企业组织架构功能开发（参考钉钉多企业隔离 · 完整版）

> 基于"增强 `accounts.Department` + `org` 应用专注业务扩展 + **多企业/集团数据隔离**"的设计原则，结合 `企联云` 当前技术栈（Django 4.2 + DRF + Channels + 原生 JS）生成完整开发提示词。

---

## 🎯 一、核心设计原则

### 1.1 三大铁律

| 原则                      | 说明                                                         |
| ------------------------- | ------------------------------------------------------------ |
| **不重新定义 Department** | 增强 `accounts.Department`，`org` 应用只做业务扩展，避免外键断裂和 Admin 紊乱 |
| **多企业行级隔离**        | 所有业务表通过 `tenant_id` 字段隔离，共享数据库，查询自动过滤 |
| **用户可跨企业**          | 一个用户可加入多个企业/集团，参考钉钉的组织切换体验          |

### 1.2 多租户隔离架构总览

```
┌──────────────────────────────────────────────────────────────────┐
│                          用户请求层                               │
│  ┌──────────────┐  ┌──────────────┐  ┌──────────────┐          │
│  │ 企业切换组件   │  │ JWT Token    │  │ X-Tenant-Id  │          │
│  │ (前端)        │  │ (认证)       │  │ (请求头)     │          │
│  └──────┬───────┘  └──────┬───────┘  └──────┬───────┘          │
│         └────────────┬────┘──────────────────┘                   │
│                      ▼                                           │
│  ┌───────────────────────────────────────────────────┐           │
│  │          TenantMiddleware（多租户中间件）            │           │
│  │  1. 解析用户当前激活的企业（active_tenant）          │           │
│  │  2. 将 tenant 挂载到 request.tenant                │           │
│  │  3. 激活 TenantFilter（自动过滤查询）               │           │
│  └───────────────────────┬───────────────────────────┘           │
│                          ▼                                       │
│  ┌───────────────────────────────────────────────────┐           │
│  │               数据访问层（自动隔离）                 │           │
│  │  • Department.objects.all() → 自动加 tenant 过滤   │           │
│  │  • ChatRoom.objects.all() → 自动加 tenant 过滤     │           │
│  │  • 文件/审批/任务 → 自动加 tenant 过滤              │           │
│  └───────────────────────────────────────────────────┘           │
└──────────────────────────────────────────────────────────────────┘
```

### 1.3 企业/集团层级模型

```python
class Tenant(models.Model):
    """企业/租户模型（多租户隔离的核心）"""
    
    TENANT_TYPE_CHOICES = [
        ('group', '集团'),
        ('company', '公司'),
        ('branch', '分公司'),
        ('virtual', '虚拟组织'),
    ]
    
    # ===== 基础信息 =====
    name = models.CharField(max_length=200, verbose_name='企业名称')
    short_name = models.CharField(max_length=50, blank=True, verbose_name='企业简称')
    code = models.CharField(max_length=50, unique=True, verbose_name='企业编码',
                            help_text='全局唯一，如：ALIBABA、TENCENT')
    logo = models.ImageField(upload_to='tenants/logos/', blank=True, verbose_name='企业Logo')
    
    # ===== 集团层级 =====
    parent = models.ForeignKey('self', null=True, blank=True,
                               on_delete=models.SET_NULL,
                               related_name='sub_tenants',
                               verbose_name='上级企业/集团')
    tenant_type = models.CharField(max_length=20, choices=TENANT_TYPE_CHOICES,
                                   default='company', verbose_name='企业类型')
    level = models.IntegerField(default=1, verbose_name='层级')
    
    # ===== 企业信息 =====
    industry = models.CharField(max_length=100, blank=True, verbose_name='所属行业')
    scale = models.CharField(max_length=50, blank=True, verbose_name='企业规模',
                             help_text='如：1-50人、50-200人、200人以上')
    address = models.CharField(max_length=500, blank=True, verbose_name='企业地址')
    contact_phone = models.CharField(max_length=20, blank=True, verbose_name='联系电话')
    website = models.URLField(blank=True, verbose_name='企业官网')
    description = models.TextField(blank=True, verbose_name='企业简介')
    
    # ===== 超级管理员 =====
    owner = models.ForeignKey(
        settings.AUTH_USER_MODEL,
        on_delete=models.SET_NULL,
        null=True,
        related_name='owned_tenants',
        verbose_name='企业所有者/超级管理员'
    )
    
    # ===== 配置与限制 =====
    max_users = models.IntegerField(default=100, verbose_name='最大用户数')
    max_storage_gb = models.IntegerField(default=10, verbose_name='最大存储空间(GB)')
    features = models.JSONField(default=dict, blank=True, verbose_name='功能开关',
                                help_text='如：{"org_chart": true, "approval": true}')
    custom_domain = models.CharField(max_length=200, blank=True, verbose_name='自定义域名')
    
    # ===== 状态 =====
    is_active = models.BooleanField(default=True, verbose_name='是否启用')
    is_verified = models.BooleanField(default=False, verbose_name='是否已认证')
    
    # ===== 审计字段 =====
    created_at = models.DateTimeField(auto_now_add=True, verbose_name='创建时间')
    updated_at = models.DateTimeField(auto_now=True, verbose_name='更新时间')
    
    class Meta:
        verbose_name = '企业/租户'
        verbose_name_plural = '企业/租户'
        ordering = ['name']
        indexes = [
            models.Index(fields=['code']),
            models.Index(fields=['parent', 'is_active']),
            models.Index(fields=['tenant_type']),
        ]
    
    def __str__(self):
        return self.name
    
    def save(self, *args, **kwargs):
        if self.parent:
            self.level = self.parent.level + 1
        else:
            self.level = 1
        super().save(*args, **kwargs)
    
    def get_all_sub_tenant_ids(self):
        """获取所有子企业ID（递归）"""
        ids = [self.id]
        for sub in self.sub_tenants.filter(is_active=True):
            ids.extend(sub.get_all_sub_tenant_ids())
        return ids
    
    def get_root_tenant(self):
        """获取集团根节点"""
        current = self
        while current.parent:
            current = current.parent
        return current
```

```python
class CustomUser(AbstractUser):
    """扩展用户模型（支持多企业）"""
    
    # ... 保留原有所有字段 ...
    
    # ===== 🔑 新增：当前激活的企业 =====
    active_tenant = models.ForeignKey(
        Tenant,
        on_delete=models.SET_NULL,
        null=True,
        blank=True,
        related_name='active_users',
        verbose_name='当前激活企业'
    )
    
    # ===== 新增方法（多企业支持） =====
    def get_tenants(self):
        """获取用户所属的所有企业"""
        return Tenant.objects.filter(
            memberships__user=self,
            memberships__is_active=True
        )
    
    def get_active_tenant(self):
        """获取当前激活的企业"""
        if self.active_tenant:
            return self.active_tenant
        # 降级：返回默认企业或第一个企业
        default_membership = self.tenant_memberships.filter(
            is_active=True, is_default=True
        ).first()
        if default_membership:
            return default_membership.tenant
        first_membership = self.tenant_memberships.filter(is_active=True).first()
        return first_membership.tenant if first_membership else None
    
    def get_tenant_role(self, tenant=None):
        """获取在指定企业中的角色"""
        tenant = tenant or self.get_active_tenant()
        if not tenant:
            return None
        membership = self.tenant_memberships.filter(tenant=tenant).first()
        return membership.role if membership else None
    
    def is_tenant_admin(self, tenant=None):
        """检查是否是企业管理员"""
        role = self.get_tenant_role(tenant)
        return role in ['owner', 'admin']
    
    def is_tenant_owner(self, tenant=None):
        """检查是否是企业所有者"""
        return self.get_tenant_role(tenant) == 'owner'
    
    # ===== 新增方法（多部门支持） =====
    def get_primary_department(self, tenant=None):
        """获取在指定企业中的主部门"""
        tenant = tenant or self.get_active_tenant()
        try:
            from org.models import UserDepartment
            primary_rel = UserDepartment.objects.filter(
                user=self,
                is_primary=True,
                department__tenant=tenant
            ).first()
            if primary_rel:
                return primary_rel.department
        except Exception:
            pass
        return self.department  # 降级到原有字段
    
    def get_all_departments(self, tenant=None):
        """获取在指定企业中的所有部门"""
        tenant = tenant or self.get_active_tenant()
        try:
            from org.models import UserDepartment
            dept_ids = UserDepartment.objects.filter(
                user=self,
                department__tenant=tenant
            ).values_list('department_id', flat=True)
            return Department.objects.filter(id__in=dept_ids)
        except Exception:
            return Department.objects.none()
    
    def switch_tenant(self, tenant_id):
        """切换激活企业"""
        tenant = Tenant.objects.get(id=tenant_id)
        # 验证用户是否属于该企业
        if not self.tenant_memberships.filter(tenant=tenant, is_active=True).exists():
            raise PermissionError('您不属于该企业')
        self.active_tenant = tenant
        self.save(update_fields=['active_tenant'])
        return tenant
```

### 3.1 TenantManager（自定义 QuerySet Manager）

```python
# accounts/managers.py
from django.db import models

class TenantQuerySet(models.QuerySet):
    """自动过滤当前企业数据的 QuerySet"""
    
    def for_tenant(self, tenant):
        """按企业过滤"""
        if tenant is None:
            return self.none()
        return self.filter(tenant=tenant)
    
    def for_user(self, user):
        """按用户当前激活企业过滤"""
        tenant = user.get_active_tenant()
        return self.for_tenant(tenant)


class TenantManager(models.Manager):
    """自动隔离的 Manager"""
    
    def get_queryset(self):
```

```python
# accounts/middleware.py
import threading
from django.utils.deprecation import MiddlewareMixin
from .models import Tenant

# 🔑 线程局部变量：存储当前请求的 tenant
_thread_locals = threading.local()

def get_current_tenant():
    """获取当前线程的 tenant"""
    return getattr(_thread_locals, 'tenant', None)

def set_current_tenant(tenant):
    """设置当前线程的 tenant"""
    _thread_locals.tenant = tenant


class TenantMiddleware(MiddlewareMixin):
    """多租户中间件：自动识别并挂载当前企业"""
    
    def process_request(self, request):
        # 1. 未登录用户不设 tenant
        if not hasattr(request, 'user') or request.user.is_anonymous:
            request.tenant = None
            set_current_tenant(None)
            return
        
        # 2. 优先从请求头获取（前端切换企业时传递）
        tenant_id = request.headers.get('X-Tenant-Id')
        if tenant_id:
            try:
                tenant = Tenant.objects.get(id=tenant_id, is_active=True)
                # 验证用户是否属于该企业
                if request.user.tenant_memberships.filter(
                    tenant=tenant, is_active=True
                ).exists():
                    request.tenant = tenant
                    set_current_tenant(tenant)
                    return
            except Tenant.DoesNotExist:
                pass
        
        # 3. 降级：使用用户激活的企业
        request.tenant = request.user.get_active_tenant()
        set_current_tenant(request.tenant)
    
    def process_response(self, request, response):
        # 清理线程局部变量
        set_current_tenant(None)
        return response
```

```python
# org/filters.py
from rest_framework.filters import BaseFilterBackend

class TenantFilterBackend(BaseFilterBackend):
    """DRF 过滤器：自动按当前企业过滤数据"""
    
    def filter_queryset(self, request, queryset, view):
        tenant = getattr(request, 'tenant', None)
        if tenant is None:
            return queryset.none()
        
        # 如果 queryset 的模型有 tenant 字段，自动过滤
        if hasattr(queryset.model, 'tenant'):
            return queryset.filter(tenant=tenant)
        
        # 特殊处理：通过关联字段过滤
        tenant_field = getattr(view, 'tenant_field', None)
        if tenant_field:
            return queryset.filter(**{tenant_field: tenant})
        
        return queryset
```

```python
            'member_count': department.get_member_count(),
            'children': []
        }
        if max_depth > 0:
            children = Department.objects.filter(
                parent=department, is_active=True
            ).order_by('sort_order')
            for child in children:
                data['children'].append(self._build_org_chart(child, max_depth - 1))
        return data
```

```python
# org/permissions.py
from rest_framework import permissions

class OrgPermission(permissions.BasePermission):
    """组织架构企业级权限控制"""
    
    def has_permission(self, request, view):
        user = request.user
        tenant = getattr(request, 'tenant', None)
        
        if not tenant:
            return False
        
        # 🔑 首先检查用户是否属于当前企业
        if not user.tenant_memberships.filter(tenant=tenant, is_active=True).exists():
            return False
        
        role = user.get_tenant_role(tenant)
        
        # 企业所有者/管理员：完整权限
        if role in ['owner', 'admin']:
            return True
        
        # 部门管理员：本部门及子部门
        if role == 'dept_admin':
            return self._can_manage_dept(user, view, tenant)
        
        # 普通成员：只读公开部门
        if view.action in ['list', 'retrieve', 'org_chart']:
            return True
        
        return False
    
    def _can_manage_dept(self, user, view, tenant):
        """检查部门管理员权限"""
        managed_dept_ids = []
        for dept in user.managed_departments.filter(tenant=tenant):
            managed_dept_ids.append(dept.id)
            managed_dept_ids.extend(dept.get_descendant_ids())
        # 存入 request 供 view 使用
        request._managed_dept_ids = managed_dept_ids
        return view.action in ['list', 'retrieve', 'members', 'update', 'partial_update']
```

```python
# Step 1: 创建默认企业
def create_default_tenant(apps, schema_editor):
    Tenant = apps.get_model('accounts', 'Tenant')
    CustomUser = apps.get_model('accounts', 'CustomUser')
    
    # 创建默认企业
    default_tenant = Tenant.objects.create(
        name='默认企业',
        code='DEFAULT',
        tenant_type='company',
        is_active=True
    )
    
    # Step 2: 将所有现有用户关联到默认企业
    for user in CustomUser.objects.all():
        TenantMembership.objects.create(
            user=user,
            tenant=default_tenant,
            role='admin' if user.is_staff else 'member',
            is_active=True,
            is_default=True
        )
        user.active_tenant = default_tenant
        user.save(update_fields=['active_tenant'])
    
    # Step 3: 将所有现有部门关联到默认企业
    Department = apps.get_model('accounts', 'Department')
    for dept in Department.objects.all():
        dept.tenant = default_tenant
        dept.save(update_fields=['tenant'])
    
    # Step 4: 迁移用户部门关系
    UserDepartment = apps.get_model('org', 'UserDepartment')
    for user in CustomUser.objects.filter(department__isnull=False):
        UserDepartment.objects.get_or_create(
            user=user,
            department=user.department,
            defaults={'is_primary': True, 'position': user.position or ''}
        )
    
    # Step 5: 将聊天室关联到默认企业
    ChatRoom = apps.get_model('chat', 'ChatRoom')
    ChatRoom.objects.filter(tenant__isnull=True).update(tenant=default_tenant)
```

## ✅ 十二、验收标准

- [ ] 用户可创建/加入多个企业
- [ ] 用户可在企业间切换，切换后所有数据自动过滤
- [ ] 每个企业有独立的组织架构树，互不可见
- [ ] 部门 CRUD 自动关联当前企业
- [ ] 企业管理员只能管理本企业数据
- [ ] 集团管理员可查看子企业数据
- [ ] 部门群自动创建并同步成员
- [ ] 组织架构图正确渲染
- [ ] PC 端和移动端均适配
- [ ] 现有聊天、网盘、OA 功能不受影响（数据隔离后正常工作）
- [ ] API 日志正确记录 org 相关操作
